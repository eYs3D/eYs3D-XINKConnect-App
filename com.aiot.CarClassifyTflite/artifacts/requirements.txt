# Inference runtime. Board (aarch64 Vivante NPU) installs tflite-runtime and runs
# via the VX delegate. x86_64 desktops have NO tflite-runtime wheel, so install
# ai-edge-litert there instead (official lightweight LiteRT, ~17 MB, inference-
# only): backend/tflite_backend.py imports it via the same interpreter API and
# uses the CPU interpreter when libvx_delegate.so is absent.
# The aarch64 line is unchanged, so the board install behaves exactly as before.
tflite-runtime>=2.13.0; platform_machine == "aarch64" and sys_platform == "linux"
ai-edge-litert==1.2.0; platform_machine == "x86_64" or sys_platform == "darwin"

# 8.0.238, AGPL-3.0). It needs scipy for the Kalman filter and linear assignment.
# (ultralytics itself is NOT installed — its `lap` dep has no py3.8/aarch64 wheel.)
scipy>=1.8.0
